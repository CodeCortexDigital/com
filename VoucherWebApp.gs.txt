/**
 * CODE CORTEX - SINGLE-VOUCHER PRINT / PDF (Google Apps Script web app)
 *
 * What it does: opens ONE voucher (the row you clicked) in the Code Cortex voucher format, with two buttons:
 *   [Print]  and  [Download PDF]  (the PDF is also saved in your Drive folder "Code Cortex Vouchers").
 *
 * Setup (once):
 * 1) Upload the Excel workbook to Google Drive, open it with Google Sheets (File > Save as Google Sheets).
 * 2) Extensions > Apps Script. Delete the old code, paste this whole file as Code.gs. Save.
 * 3) Run the function authorize() once and allow access (Drive + Sheets).
 * 4) Deploy > New deployment > Web app > Execute as: Me > Who has access: Only myself > Deploy. Copy the URL.
 * 5) In the sheet, paste the URL into Assumptions!C32. Done.
 *    Voucher Register column X = "Print voucher", column AA = "Download PDF" (one click each, one voucher only).
 *    After changing this code, use Deploy > Manage deployments > Edit > New version (the URL stays the same).
 *
 * Columns used on 'Voucher Register' (row = ?row=N):
 *  B date, C voucher no, D type, E party, F particulars, G Dr code, H Dr account, I funding source,
 *  J Cr code, K Cr account, L amount, M payment method, N supporting document, O bank rec, P prepared by,
 *  Q verified by, Z printed narration.
 */
var LOGO_URL = '';   // optional: public https URL of the company logo (leave blank to omit)
var SHEET_NAME = 'Voucher Register';
var FOLDER_NAME = 'Code Cortex Vouchers';
var CO = 'Code Cortex (SMC) (Private) Limited';
var NTN = 'H223725', CUIN = '0303684';
var BANNER = 'Audit Compliance: SECP Companies Act 2017 (Sec 237) | FBR Income Tax Ordinance 2001 (Sec 174)';
var LINES = [
  'SECP Compliance: Companies Act 2017 - Section 237 (Books of Accounts)',
  'FBR Compliance: Income Tax Ordinance 2001 - Section 174 (Record Keeping)',
  'Sales Tax Act 1990: Section 26 (Tax Invoice Requirements)',
  'Retention Period: Minimum 6 years as per legal requirements',
  'Audit Ready: Supporting documents retained for SECP/FBR inspection'];
var FOOT = [
  'This is a system-controlled financial record of ' + CO + '.',
  'Generated for internal audit, SECP inspection, and FBR tax audit purposes.',
  'Prepared from books of accounts maintained under Section 237 of Companies Act 2017.'];

function authorize() { DriveApp.getRootFolder(); SpreadsheetApp.getActive(); }

function doGet(e) {
  var row = parseInt(e.parameter.row, 10);
  if (!row || row < 5) return HtmlService.createHtmlOutput('Invalid row');
  var d = readVoucher(row);
  if (!d) return HtmlService.createHtmlOutput('No voucher on this row');
  if (e.parameter.pdf) {
    var blob = HtmlService.createHtmlOutput(voucherHtml(d, false)).getBlob().getAs('application/pdf').setName('Voucher-' + d.no + '.pdf');
    var folder = getFolder();
    var old = folder.getFilesByName(blob.getName());
    while (old.hasNext()) old.next().setTrashed(true);
    var file = folder.createFile(blob);
    var url = 'https://drive.google.com/uc?export=download&id=' + file.getId();
    return HtmlService.createHtmlOutput(
      '<p style="font-family:Arial">Downloading <b>' + d.no + '.pdf</b> ... if it does not start, <a target="_top" href="' + url + '">click here</a>. ' +
      'A copy is saved in your Drive folder "' + FOLDER_NAME + '".</p>' +
      '<script>window.top.location.href="' + url + '";</script>').setTitle('Voucher PDF - ' + d.no);
  }
  var self = ScriptApp.getService().getUrl() + '?row=' + row;
  var bar = '<div class="np" style="text-align:right;margin-bottom:10px">' +
    '<button onclick="window.print()" style="padding:8px 16px;font-size:14px">Print</button> ' +
    '<a target="_top" href="' + self + '&pdf=1"><button style="padding:8px 16px;font-size:14px">Download PDF</button></a></div>';
  return HtmlService.createHtmlOutput(voucherHtml(d, true, bar)).setTitle('Voucher - ' + d.no);
}

function getFolder() {
  var it = DriveApp.getFoldersByName(FOLDER_NAME);
  return it.hasNext() ? it.next() : DriveApp.createFolder(FOLDER_NAME);
}

function readVoucher(row) {
  var sh = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME);
  var v = sh.getRange(row, 1, 1, 26).getDisplayValues()[0];
  var raw = sh.getRange(row, 1, 1, 26).getValues()[0];
  if (!v[2]) return null;
  var tz = Session.getScriptTimeZone();
  return {
    no: v[2], type: v[3], party: v[4], part: v[5], drCode: v[6], drAcc: v[7], crCode: v[9], crAcc: v[10],
    amount: Number(raw[11]) || 0, method: v[12], support: v[13], rec: v[14], prep: v[15], ver: v[16], narr: v[25],
    date: Utilities.formatDate(new Date(raw[1]), tz, 'dd-MMM-yyyy'),
    now: Utilities.formatDate(new Date(), tz, 'dd-MMM-yyyy hh:mm a')
  };
}

function esc(s) { return String(s == null ? '' : s).replace(/&/g, '&amp;').replace(/</g, '&lt;'); }

function voucherHtml(d, auto, bar) {
  var vtype = d.no.indexOf('RV') === 0 ? 'Receipt Voucher (RV)' : d.no.indexOf('PV') === 0 ? 'Payment Voucher (PV)' : d.no.indexOf('JV') === 0 ? 'Journal Voucher (JV)' : 'Voucher';
  var entry = (d.type === 'Contra' ? 'Contra Entry' : d.type + ' Entry') + '<br>Dr ' + esc(d.drCode) + ' ' + esc(d.drAcc) + '<br>Cr ' + esc(d.crCode) + ' ' + esc(d.crAcc);
  var rec = d.rec === 'Y' ? 'YES' : d.rec === 'N' ? 'NO' : d.rec;
  var L = 'style="border:1.5px solid #000;padding:9px 10px;font-weight:bold;color:#1f3a8a;width:24%"';
  var C = 'style="border:1.5px solid #000;padding:9px 10px"';
  var G = 'style="border:1.5px solid #000;padding:12px 8px;text-align:center;font-weight:bold;color:#1b5e20"';
  var H = 'style="border:1.5px solid #000;padding:8px;text-align:center;color:#1f3a8a"';
  var T = 'width="100%" cellspacing="0" cellpadding="0" style="border-collapse:collapse;margin-bottom:14px;font-size:13px"';
  return '<html><head><title>Voucher - ' + esc(d.no) + '</title><style>' +
    '@page{size:A4;margin:12mm}body{font-family:Arial,Helvetica,sans-serif;margin:0 auto;width:740px;color:#000}' +
    '@media print{.np{display:none}}</style></head><body>' + (bar || '') +
    '<table width="100%" cellspacing="0" cellpadding="0" style="border-bottom:3px solid #1f3a8a;margin-bottom:12px"><tr>' +
    '<td width="100">' + (LOGO_URL ? '<img src="' + LOGO_URL + '" width="80" height="80">' : '') + '</td>' +
    '<td align="center" style="color:#1f3a8a"><div style="font-size:26px;font-weight:bold">' + esc(CO) + '</div>' +
    '<div style="font-size:13px;font-weight:bold;margin-top:4px">' + vtype + ' (SECP &amp; FBR Compliant)</div>' +
    '<div style="font-size:11px;color:#555">NTN: ' + NTN + ' | CUIN: ' + CUIN + '</div></td><td width="100"></td></tr></table>' +
    '<div style="border-left:4px solid #1f3a8a;padding:6px 10px;margin:14px 0;font-size:12px;font-weight:bold">&#128274; ' + esc(BANNER) + '</div>' +
    '<table ' + T + '><tr><td ' + L + '>Voucher Type</td><td ' + C + '>' + vtype + '</td><td ' + L + '>Voucher No</td><td ' + C + '>' + esc(d.no) + '</td></tr>' +
    '<tr><td ' + L + '>Transaction Date</td><td ' + C + '>' + d.date + '</td><td ' + L + '>Party / Payee</td><td ' + C + '>' + esc(d.party) + '</td></tr></table>' +
    '<table ' + T + '><tr><td ' + L + '>Description / Purpose</td><td ' + C + '>' + esc(d.part) + '</td></tr>' +
    '<tr><td ' + L + '>Category (COA Head)</td><td ' + C + '>' + esc(d.drAcc) + '</td></tr>' +
    '<tr><td ' + L + '>COA Code</td><td ' + C + '>' + esc(d.drCode) + '</td></tr>' +
    '<tr><td ' + L + '>Payment Mode</td><td ' + C + '>' + esc(d.method) + '</td></tr></table>' +
    '<table ' + T + '><tr><td ' + H + '><b>Entry Type</b></td><td ' + H + '><b>Amount (PKR)</b></td><td ' + H + '><b>Amount in Words</b></td></tr>' +
    '<tr><td ' + G + '>' + entry + '</td><td ' + G + '>PKR ' + Number(d.amount).toLocaleString('en-US') + '</td><td ' + G + '>' + amountWords(d.amount) + '</td></tr></table>' +
    '<table ' + T + '><tr><td ' + L + '>Narration / Remarks</td><td ' + C + '>' + esc(d.narr) + '</td></tr>' +
    '<tr><td ' + L + '>Supporting Document</td><td ' + C + '>' + esc(d.support) + '</td></tr>' +
    '<tr><td ' + L + '>Bank Reconciliation</td><td ' + C + '>' + esc(rec) + '</td></tr>' +
    '<tr><td ' + L + '>Posting Status</td><td ' + C + '>POSTED</td></tr>' +
    '<tr><td ' + L + '>Printed On</td><td ' + C + '>' + d.now + '</td></tr></table>' +
    '<div style="border-left:4px solid #0b3d5c;padding:6px 10px;font-size:11.5px;line-height:1.6">' +
    LINES.map(function (x) { return '&#10004; ' + esc(x); }).join('<br>') + '</div>' +
    '<table width="100%" cellspacing="12" style="margin-top:30px;font-size:12px;text-align:center"><tr>' +
    '<td style="border-top:2px solid #000"><b>Prepared By</b><br>' + esc(d.prep) + '<br><span style="font-size:10px">Signature &amp; Date</span></td>' +
    '<td style="border-top:2px solid #000"><b>Verified By</b><br>' + esc(d.ver) + '<br><span style="font-size:10px">Signature &amp; Date</span></td>' +
    '<td style="border-top:2px solid #000"><b>CEO / Director</b><br>&nbsp;<br><span style="font-size:10px">Signature &amp; Company Stamp</span></td></tr></table>' +
    '<div style="border-top:1px dashed #999;margin-top:14px;padding-top:8px;text-align:center;font-size:10px;color:#555">' +
    FOOT.map(esc).join('<br>') + '<br><b>Document ID: ' + esc(d.no) + ' | Printed: ' + d.now + '</b></div>' +
    (auto ? '<script>window.onload=function(){}</script>' : '') + '</body></html>';
}

var ONES = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten', 'Eleven', 'Twelve', 'Thirteen',
  'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen'];
var TENS = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety'];
function w99(n) { return n < 20 ? ONES[n] : TENS[Math.floor(n / 10)] + (n % 10 ? ' ' + ONES[n % 10] : ''); }
function amountWords(n) {
  n = Math.floor(n);
  var s = 'Pakistan Rupees ';
  var lakh = Math.floor(n / 100000), th = Math.floor((n % 100000) / 1000), h = Math.floor((n % 1000) / 100), r = n % 100;
  if (lakh) s += w99(lakh) + ' Lakh ';
  if (th) s += w99(th) + ' Thousand ';
  if (h) s += w99(h) + ' Hundred ';
  if (r) s += w99(r) + ' ';
  return s + 'Only';
}
